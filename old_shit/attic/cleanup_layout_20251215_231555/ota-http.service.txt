# /etc/systemd/system/ota-http.service
#!/usr/bin/python3
[Unit]
Description=OTA Firmware HTTP Server
After=network.target

[Service]
Type=simple
User=rudyy
Group=rudyy
WorkingDirectory=/home/rudyy/er1
ExecStartPre=/usr/bin/mkdir -p /home/rudyy/firmware
ExecStart=/usr/bin/python3 /home/rudyy/er1/scripts/ota_http.py daemon
Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/ota-http.service.d/override.conf
[Service]
AmbientCapabilities=CAP_NET_BIND_SERVICE
CapabilityBoundingSet=CAP_NET_BIND_SERVICE
