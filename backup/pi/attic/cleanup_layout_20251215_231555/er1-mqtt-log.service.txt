# /etc/systemd/system/er1-mqtt-log.service
[Unit]
Description=ER1 MQTT logger (+/log,+/hb,+/evt,+/state -> /home/rudyy/er1/logs/er1-DD.MM.YYYY.log)
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=rudyy
Group=rudyy
WorkingDirectory=/home/rudyy/er1
Environment=LOCAL_BROKER=127.0.0.1
ExecStart=/home/rudyy/er1/scripts/mqtt_logs.sh daemon
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
